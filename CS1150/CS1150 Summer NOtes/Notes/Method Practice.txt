import java.util.*;

public class Main {
    public static void main(String[] args) {
        int y = intDoubler(10);

        System.out.println(y);
        System.out.println(sum(1, 10));
        printDisplay();
        return;
        }
        //doubles the integer value and returns it
        public static int intDoubler(int x){
            return x*2;
        }
        public static int sum(int sumCount){
            int result = 0;
            for(int i = 0; i < sumCount; i++){
                result += i;
            }
            return result;
        }
        public static int sum(int i1, int i2) {
            int sum = 0;
            for (int i = i1; i <= i2; i++){
                sum += i;
                System.out.println("Sum is: " + sum + " and the i value is: " + i);
            }
            return sum;
        }
        public static void printDisplay(){
            System.out.println("Test Successful");
        }
}
